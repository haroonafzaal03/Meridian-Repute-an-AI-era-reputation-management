<?php
// contact.smtp.php — FALLBACK handler using authenticated SMTP (PHPMailer).
// Use this ONLY if mail() + SPF/DKIM/DMARC still lands in spam.
// Requires PHPMailer. On cPanel:  composer require phpmailer/phpmailer
// (or upload the PHPMailer src/ and adjust the requires below).
declare(strict_types=1);

use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

// --- Load PHPMailer (Composer autoload preferred) ---
$autoload = __DIR__ . '/vendor/autoload.php';
if (is_file($autoload)) {
    require $autoload;
} else {
    // Manual include fallback — adjust path to where you uploaded PHPMailer:
    require __DIR__ . '/PHPMailer/src/Exception.php';
    require __DIR__ . '/PHPMailer/src/PHPMailer.php';
    require __DIR__ . '/PHPMailer/src/SMTP.php';
}

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);
    exit('Method Not Allowed');
}

// --- Honeypot ---
if (!empty($_POST['website'] ?? '')) {
    http_response_code(200);
    exit('OK');
}

// --- Collect + validate (match your form field names) ---
$name    = trim($_POST['name']    ?? '');
$email   = trim($_POST['email']   ?? '');
$phone   = trim($_POST['phone']   ?? '');
$message = trim($_POST['message'] ?? '');

if ($name === '' || $message === '' || !filter_var($email, FILTER_VALIDATE_EMAIL)) {
    http_response_code(400);
    exit('Invalid submission');
}

// --- SMTP credentials.
//     Best practice: keep the password OUT of the webroot. Set these as
//     environment variables in cPanel, or in a file above public_html that
//     you include here. Hardcoding works but is less safe. ---
$SMTP_HOST = getenv('MR_SMTP_HOST') ?: 'localhost';                  // e.g. mail.meridianrepute.com
$SMTP_USER = getenv('MR_SMTP_USER') ?: 'people@meridianrepute.com';
$SMTP_PASS = getenv('MR_SMTP_PASS') ?: '';                          // set MR_SMTP_PASS in cPanel env
$SMTP_PORT = (int) (getenv('MR_SMTP_PORT') ?: 587);                  // 587 STARTTLS, or 465 SMTPS

$mail = new PHPMailer(true);
try {
    $mail->isSMTP();
    $mail->Host       = $SMTP_HOST;
    $mail->SMTPAuth   = true;
    $mail->Username   = $SMTP_USER;
    $mail->Password   = $SMTP_PASS;
    $mail->Port       = $SMTP_PORT;
    $mail->SMTPSecure = ($SMTP_PORT === 465)
        ? PHPMailer::ENCRYPTION_SMTPS
        : PHPMailer::ENCRYPTION_STARTTLS;
    $mail->CharSet    = 'UTF-8';

    // From is the authenticated domain mailbox (aligns SPF + DKIM).
    $mail->setFrom('people@meridianrepute.com', 'Meridian Repute');
    $mail->addAddress('people@meridianrepute.com');
    // Visitor goes in Reply-To, never From.
    $mail->addReplyTo($email, $name);

    $mail->Subject = 'New consultation request from ' . $name;
    $mail->isHTML(false);
    $mail->Body =
        "New consultation request\n" .
        "-------------------------\n" .
        "Name:    {$name}\n" .
        "Email:   {$email}\n" .
        "Phone:   {$phone}\n\n" .
        "Message:\n{$message}\n";

    $mail->send();
    header('Location: /thank-you.html');
    exit;
} catch (Exception $e) {
    error_log('contact.smtp.php send failed: ' . $mail->ErrorInfo);
    http_response_code(500);
    exit('Mail send failed');
}
